**Goal:** A class-conditional DDPM that produces recognizable CIFAR-10 samples after ≤1h of H100
training on Modal, plus a template + numeric-oracle test suite for hand re-implementation.

**Architecture:** One reference module `02-ddpm/model.py` holds the UNet (ResBlock, single-head
Attention at 16×16, sinusoidal timestep embedding, class embedding with a null token) and a
`Diffusion` module wrapping it (cosine ᾱ schedule, `q_sample`, `p_losses`, ancestral and DDIM
samplers, classifier-free guidance). `02-ddpm/train.py` is a Modal entrypoint. Tests import either
`model.py` or `template.py` through `utils.load_impl`, selected by the `IMPL` env var, and compare
them numerically under a fixed seed after `load_state_dict` transfer.

**Tech Stack:** Python 3.12, torch 2.13, torchvision, modal 1.5, pytest.

### Task 2: Cosine noise schedule

**Files:**
- Modify: `02-ddpm/model.py`
- Modify: `02-ddpm/tests/test_ddpm.py`

**Interfaces:**
- Consumes: `utils.load_impl`.
- Produces: `cosine_abar(T: int = 1000, s: float = 0.008) -> Tensor` of shape `(T,)`, the cumulative
  product ᾱ_t. `timestep_embedding(t: Tensor, dim: int) -> Tensor` of shape `(len(t), dim)`.

The schedule is `f(t) = cos((t/T + s)/(1 + s) · π/2)²`, `ᾱ_t = f(t)/f(0)`, then
`β_t = clamp(1 − ᾱ_t/ᾱ_{t−1}, max=0.999)` and ᾱ is recomputed as `cumprod(1 − β)`. The clamp is
why we recompute rather than return `f(t)/f(0)` directly.

```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F


def cosine_abar(T=1000, s=0.008):
    t = torch.linspace(0, T, T + 1) / T
    f = torch.cos((t + s) / (1 + s) * math.pi / 2) ** 2
    abar = f / f[0]
    betas = (1 - abar[1:] / abar[:-1]).clamp(max=0.999)
    return torch.cumprod(1 - betas, dim=0)


def timestep_embedding(t, dim):
    half = dim // 2
    freqs = torch.exp(-math.log(10000) * torch.arange(half, device=t.device) / half)
    args = t[:, None].float() * freqs[None]
    return torch.cat([torch.cos(args), torch.sin(args)], dim=-1)
```

a shape test.

```python
class ResBlock(nn.Module):
    def __init__(self, in_ch, out_ch, emb_dim):
        super().__init__()
        self.norm1 = nn.GroupNorm(32, in_ch)
        self.conv1 = nn.Conv2d(in_ch, out_ch, 3, padding=1)
        self.emb = nn.Linear(emb_dim, out_ch)
        self.norm2 = nn.GroupNorm(32, out_ch)
        self.conv2 = nn.Conv2d(out_ch, out_ch, 3, padding=1)
        self.skip = nn.Conv2d(in_ch, out_ch, 1) if in_ch != out_ch else nn.Identity()

    def forward(self, x, emb):
        h = self.conv1(F.silu(self.norm1(x)))
        h = h + self.emb(F.silu(emb))[:, :, None, None]
        h = self.conv2(F.silu(self.norm2(h)))
        return h + self.skip(x)


class Attention(nn.Module):
    def __init__(self, ch):
        super().__init__()
        self.norm = nn.GroupNorm(32, ch)
        self.qkv = nn.Conv2d(ch, ch * 3, 1)
        self.proj = nn.Conv2d(ch, ch, 1)

    def forward(self, x):
        b, c, h, w = x.shape
        qkv = self.qkv(self.norm(x)).reshape(b, 3, c, h * w).transpose(2, 3)
        q, k, v = qkv.unbind(1)
        out = F.scaled_dot_product_attention(q, k, v)
        return x + self.proj(out.transpose(1, 2).reshape(b, c, h, w))


class Downsample(nn.Module):
    def __init__(self, ch):
        super().__init__()
        self.conv = nn.Conv2d(ch, ch, 3, stride=2, padding=1)

    def forward(self, x):
        return self.conv(x)


class Upsample(nn.Module):
    def __init__(self, ch):
        super().__init__()
        self.conv = nn.Conv2d(ch, ch, 3, padding=1)

    def forward(self, x):
        return self.conv(F.interpolate(x, scale_factor=2, mode="nearest"))
```

```python
class UNet(nn.Module):
    def __init__(self, ch=128, mults=(1, 2, 2, 2), n_classes=10, emb_dim=512, attn_res=16):
        super().__init__()
        self.ch = ch
        self.time_mlp = nn.Sequential(nn.Linear(ch, emb_dim), nn.SiLU(), nn.Linear(emb_dim, emb_dim))
        self.label_emb = nn.Embedding(n_classes + 1, emb_dim)
        self.conv_in = nn.Conv2d(3, ch, 3, padding=1)
        self.down, skips, cur, res = nn.ModuleList(), [ch], ch, 32
        for i, m in enumerate(mults):
            for _ in range(2):
                stage = nn.ModuleList([ResBlock(cur, ch * m, emb_dim)])
                cur = ch * m
                if res == attn_res:
                    stage.append(Attention(cur))
                self.down.append(stage)
                skips.append(cur)
            if i < len(mults) - 1:
                self.down.append(nn.ModuleList([Downsample(cur)]))
                skips.append(cur)
                res //= 2
        self.mid = nn.ModuleList([ResBlock(cur, cur, emb_dim), Attention(cur), ResBlock(cur, cur, emb_dim)])
        self.up = nn.ModuleList()
        for i, m in reversed(list(enumerate(mults))):
            for _ in range(3):
                stage = nn.ModuleList([ResBlock(cur + skips.pop(), ch * m, emb_dim)])
                cur = ch * m
                if res == attn_res:
                    stage.append(Attention(cur))
                self.up.append(stage)
            if i > 0:
                self.up.append(nn.ModuleList([Upsample(cur)]))
                res *= 2
        self.norm_out = nn.GroupNorm(32, cur)
        self.conv_out = nn.Conv2d(cur, 3, 3, padding=1)

    def forward(self, x, t, y):
        emb = self.time_mlp(timestep_embedding(t, self.ch)) + self.label_emb(y)
        h = self.conv_in(x)
        skips = [h]
        for stage in self.down:
            h = self.run_stage(stage, h, emb)
            skips.append(h)
        for block in self.mid:
            h = self.run_stage(nn.ModuleList([block]), h, emb)
        for stage in self.up:
            if isinstance(stage[0], Upsample):
                h = stage[0](h)
                continue
            h = self.run_stage(stage, torch.cat([h, skips.pop()], dim=1), emb)
        return self.conv_out(F.silu(self.norm_out(h)))

    def run_stage(self, stage, h, emb):
        for block in stage:
            h = block(h, emb) if isinstance(block, ResBlock) else block(h)
        return h
```

Note the `up` loop pops one skip per `ResBlock` and there are 3 per level (2 down blocks + the
level's input), matching the standard DDPM topology.

  - `q_sample(x0, t, noise) -> Tensor`
  - `p_losses(x0, y, p_drop=0.1) -> Tensor` (scalar)
  - `eps(x, t, y, w) -> Tensor`
  - `ddim_sample(y, steps=50, w=3.0) -> Tensor[B,3,32,32]`
  - `ddpm_sample(y, w=3.0) -> Tensor[B,3,32,32]`

Guidance convention: `eps = eps_uncond + w · (eps_cond − eps_uncond)`, so `w=1` is plain
conditional and `w=3` is the usual CIFAR setting. Record this — the other common convention
(`(1+w)·cond − w·uncond`) shifts the meaning of `w` by one.

```python
class Diffusion(nn.Module):
    def __init__(self, model, T=1000, n_classes=10):
        super().__init__()
        self.model, self.T, self.null = model, T, n_classes
        self.register_buffer("abar", cosine_abar(T))

    def q_sample(self, x0, t, noise):
        a = self.abar[t][:, None, None, None]
        return a.sqrt() * x0 + (1 - a).sqrt() * noise

    def p_losses(self, x0, y, p_drop=0.1):
        t = torch.randint(0, self.T, (x0.shape[0],), device=x0.device)
        noise = torch.randn_like(x0)
        y = torch.where(torch.rand(y.shape, device=y.device) < p_drop, self.null, y)
        return F.mse_loss(self.model(self.q_sample(x0, t, noise), t, y), noise)

    def eps(self, x, t, y, w):
        cond, uncond = self.model(
            torch.cat([x, x]), torch.cat([t, t]),
            torch.cat([y, torch.full_like(y, self.null)]),
        ).chunk(2)
        return uncond + w * (cond - uncond)

    @torch.no_grad()
    def ddim_sample(self, y, steps=50, w=3.0):
        x = torch.randn(len(y), 3, 32, 32, device=y.device)
        ts = torch.linspace(self.T - 1, 0, steps).long().to(y.device)
        for i, t in enumerate(ts):
            e = self.eps(x, t.repeat(len(y)), y, w)
            a = self.abar[t]
            a_prev = self.abar[ts[i + 1]] if i + 1 < steps else torch.ones((), device=y.device)
            x0 = ((x - (1 - a).sqrt() * e) / a.sqrt()).clamp(-1, 1)
            x = a_prev.sqrt() * x0 + (1 - a_prev).sqrt() * e
        return x

    @torch.no_grad()
    def ddpm_sample(self, y, w=3.0):
        x = torch.randn(len(y), 3, 32, 32, device=y.device)
        for t in reversed(range(self.T)):
            e = self.eps(x, torch.full((len(y),), t, device=y.device), y, w)
            a, a_prev = self.abar[t], self.abar[t - 1] if t else torch.ones((), device=y.device)
            alpha, beta = a / a_prev, 1 - a / a_prev
            mean = (x - beta / (1 - a).sqrt() * e) / alpha.sqrt()
            if t == 0:
                return mean
            var = beta * (1 - a_prev) / (1 - a)
            x = mean + var.sqrt() * torch.randn_like(x)
        return x
```

```python
def cifar_loader(batch_size):
    from torchvision import datasets, transforms
    tf = transforms.Compose([
        transforms.RandomHorizontalFlip(),
        transforms.ToTensor(),
        transforms.Normalize(0.5, 0.5),
    ])
    data = datasets.CIFAR10("/vol/data", train=True, download=True, transform=tf)
    return torch.utils.data.DataLoader(
        data, batch_size=batch_size, shuffle=True, num_workers=8, drop_last=True, pin_memory=True
    )


@torch.no_grad()
def update_ema(ema, model, decay=0.9999):
    for a, b in zip(ema.parameters(), model.parameters()):
        a.lerp_(b.detach(), 1 - decay)


def train_steps(diff, ema, loader, opt, minutes):
    import time
    start, losses, step = time.time(), [], 0
    while time.time() - start < minutes * 60:
        for x, y in loader:
            x, y = x.cuda(non_blocking=True), y.cuda(non_blocking=True)
            with torch.autocast("cuda", dtype=torch.bfloat16):
                loss = diff.p_losses(x, y)
            opt.zero_grad(set_to_none=True)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(diff.parameters(), 1.0)
            opt.step()
            update_ema(ema, diff.model)
            losses.append(loss.item())
            step += 1
            if step % 200 == 0:
                print(f"step {step} loss {sum(losses[-200:]) / 200:.4f}", flush=True)
            if time.time() - start > minutes * 60:
                break
    return losses


def save_artifacts(diff, ema, losses):
    import matplotlib.pyplot as plt
    diff.model = ema
    y = torch.arange(10, device="cuda").repeat_interleave(10)
    save_grid(diff.ddim_sample(y, steps=50, w=3.0).float().cpu(), CKPT / "samples.png", nrow=10)
    sweep = torch.cat([
        diff.ddim_sample(torch.arange(10, device="cuda"), steps=50, w=w).float().cpu()
        for w in [0.0, 1.0, 2.0, 3.0, 5.0]
    ])
    save_grid(sweep, CKPT / "cfg_sweep.png", nrow=10)
    plt.plot(torch.tensor(losses).view(-1, 50).mean(1))
    plt.xlabel("step / 50")
    plt.ylabel("loss")
    plt.savefig(CKPT / "loss.png")
    torch.save(ema.state_dict(), CKPT / "ema.pt")


@app.function(image=train_image, gpu="H100", timeout=3600, volumes={"/vol": volume})

    sys.path.insert(0, "/root/ddpm")
    from model import UNet, Diffusion
    CKPT.mkdir(parents=True, exist_ok=True)
    seed_everything(0)

    diff = Diffusion(UNet()).cuda()
    ema = copy.deepcopy(diff.model).eval().requires_grad_(False)

    opt = torch.optim.AdamW(diff.parameters(), lr=2e-4, weight_decay=0.0)

    save_artifacts(diff, ema, losses)
    volume.commit()
    print(f"done: {len(losses)} steps, final loss {sum(losses[-200:]) / 200:.4f}")


@app.local_entrypoint()
def main(minutes: float = 50.0):
    train.remote(minutes)
```

```markdown
# 02-ddpm

Denoising Diffusion Probabilistic Models (Ho et al., 2020) with the cosine schedule from
Improved DDPM (Nichol & Dhariwal, 2021), DDIM sampling (Song et al., 2021), and
classifier-free guidance (Ho & Salimans, 2022).

Class-conditional CIFAR-10, 35M-parameter UNet, ~50 minutes on one H100.

![samples](assets/samples.png)

One row per class. CFG scale w=3, 50 DDIM steps.

## The idea

Corrupt data with a known Gaussian process until it is pure noise, then learn to reverse one step
of that corruption. Because the forward process is closed-form, `q_sample` jumps to any timestep
in one shot, and the model only ever learns to predict the noise that was added.

Guidance runs the model twice per step — once with the class label, once with a null token — and
extrapolates away from the unconditional prediction. It trades diversity for fidelity.

    IMPL=template uv run pytest 02-ddpm
```
